#version 330 core


layout (location=0) in vec2 vertexPos;
layout (location=1) in vec2 frag_coord;
layout (location=2) in vec3 color;

uniform float zoom;
uniform vec2 zoom_offset;


out vec2 frag_pos;
out vec3 frag_color;




void main(){
	gl_Position = vec4(vertexPos*zoom + zoom_offset, 0.0, 1.0);

	frag_pos = frag_coord;
	frag_color = color;
	
}
